<%*
// 新建论文笔记 —— 每篇一个文件夹，笔记文件与文件夹同名(= 论文名)，与 PDF 同处一处。
// 产出与 `tools/research_wiki.py new` 完全一致的结构，规范见 wiki/AGENTS.md §3/§6.1。
// 方向列表从 papers/ 现有子文件夹读出，不写死：新建一个方向文件夹后这里立刻就能选到它。
const papers = app.vault.getAbstractFileByPath("papers");
const dirs = (papers?.children ?? [])
  .filter(f => f.children)            // 只要文件夹
  .map(f => f.name)
  .sort();
if (!dirs.length) { new Notice("papers/ 下没有方向文件夹"); return; }
const dir = await tp.system.suggester(dirs, dirs, false, "选择方向");
const paper_title = await tp.system.prompt("英文论文标题 (paper_title)");
let slug = await tp.system.prompt("文件名 slug (kebab-case)");
slug = (slug || "untitled").trim().toLowerCase().replace(/\s+/g, "-");
const year = await tp.system.prompt("年份", "2026");
const display = await tp.system.prompt("显示名 title(图谱/侧栏/索引里的论文名,英文原题;论文题目不带缩写时前置 `TIGER: `)", paper_title || slug);
const pdf = await tp.system.prompt("PDF 文件名(把它拖进本文件夹)", `【${year}】${paper_title}.pdf`);
// 必填：tags 取自 tools/tag_vocabulary.yml，2–4 个；table_summary 直接渲染成索引摘要行
const tags = await tp.system.prompt("tags(2-4 个,逗号分隔,取自 tools/tag_vocabulary.yml)", "");
const summary = await tp.system.prompt("table_summary(一句话:问题+关键方法)", "");
const org = await tp.system.prompt("organization 机构(可空)", "");
const authorsRaw = await tp.system.prompt("作者(逗号分隔完整姓名,可空)", "");
const authors = authorsRaw
  ? authorsRaw.split(/[,;，；]/).map(s => s.trim()).filter(Boolean).map(s => `"${s}"`).join(", ")
  : "";
const folder = `【${year}】${slug}`;
// 文件名 = 文件夹名，使每篇论文在图谱里是独立节点而非一堆同名 note
await tp.file.move(`/papers/${dir}/${folder}/${folder}`);
-%>
